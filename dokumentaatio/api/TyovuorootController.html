        2=>'Tiistai',
        3=>'Keskkiviikko',
        4=>'Torstai',
        5=>'Perjantai',

    $pvmstart     = $pfrom;

    $enddate    = strtotime($pto);
    $w        = $p;

      elseif($v == 4 and date('W',$startdate)%2 == 1)

      $var = 1;

          if(in_array(date('w',$startdate),$w) and $ero == $var)
